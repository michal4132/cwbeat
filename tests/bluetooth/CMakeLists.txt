cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(cwbeat_bluetooth)
target_sources(app PRIVATE src/main.c ../../subsys/bluetooth/bluetooth.c)
target_compile_definitions(app PRIVATE CONFIG_CWBEAT_BLUETOOTH_DISABLE_FEEDBACK=1 BLUETOOTH_TEST=1)
target_compile_definitions(app PRIVATE CONFIG_CWBEAT_BLUETOOTH_LOG_LEVEL=CONFIG_LOG_DEFAULT_LEVEL)
if(BT_TEST_SHUTDOWN_TIMEOUT)
 target_compile_definitions(app PRIVATE BT_TEST_SHUTDOWN_TIMEOUT=1 BLUETOOTH_SHUTDOWN_MS=295 BLUETOOTH_SHUTDOWN_PERIOD_MS=59 BLUETOOTH_SHUTDOWN_ATTEMPT_MS=49)
endif()
if(BT_TEST_COMPLETION_OVERFLOW)
 target_compile_definitions(app PRIVATE BT_TEST_COMPLETION_OVERFLOW=1)
endif()
if(BT_TEST_DISABLE_DURING_CLEANUP)
 target_compile_definitions(app PRIVATE BT_TEST_DISABLE_DURING_CLEANUP=1)
endif()
